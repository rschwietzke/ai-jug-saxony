16 package com.xceptance.xlt.report.util.rework;
17
18 import java.util.ArrayList;
19 import java.util.Arrays;
20 import java.util.List;
21
22 import com.xceptance.xlt.report.util.misc.BitCompression;
23
24 /**
25 * A {@link IntTimeSeriesEntry} stores the minimum/maximum/sum/count of all the sample values added, but can also reproduce a
26 * rough approximation of the distinct values added.
27 */
28 public class IntTimeSeriesEntry
29 {
30 // The sum for later average calculation
31 private long totalValue = 0;
32
33 // how much data have we seen
34 private int count = 0;
35
36 // how much concurrent measurements do we truly have
37 private int concurrentCount = 0;
38
39 // how many error have we seen
40 private int errorCount = 0;
41
42 // the min and max values seen
43 private int maximum = Integer.MIN_VALUE;
44 private int minimum = Integer.MAX_VALUE;
45
46 /**
47 * Holds an approximation of the distinct values added to this min-max value.
48 * This was LowPrecisionIntValueSet before. We moved it here for less memory consumption
49 * and better performance. We also limit it to 128 distinct values. We used 256 before
50 * but wee we started with XLT it was 128 for a long time.
51 */
52 private long distinctValuesLow = 0;
53 private long distinctValuesHigh = 0;
54
55 /**
56 * The value scaling factor. 2 pow scale.
57 */
58 private int distinctValuesScale = 0;
59
60 /**
61 * Constructor.
62 */
63 public IntTimeSeriesEntry()
64 {
65 }
66
67 /**
68 * Constructor.
69 *
70 * @param value
71 * the first value to add
72 */
73 public IntTimeSeriesEntry(final int firstValue, final boolean failed)
74 {
75 updateValue(firstValue, failed);
76 }
77
78 /**
79 * Adds the given sample value to this min-max value.
80 *
81 * @param value
82 * the sample to add
83 */
84 public void updateValue(final int value, final boolean failed)
85 {
86 var v = value < 0 ? 0 : value;
87
88 this.count++;
89 this.errorCount += failed ? 1 : 0;
90 this.totalValue += v;
91 this.concurrentCount++;
92
93 // use explicit comparisons to avoid Math.max/Math.min calls
94 // and writing the values when not necessary
95 if (v > this.maximum)
96 {
97 this.maximum = v;
98 }
99 if (value < this.minimum)
100 {
101 this.minimum = v;
102 }
103
104 v = scaleIfNeeded(v);
105
106 // set distinct values
107 if (v < 64)
108 {
109 distinctValuesLow |= (1L << v);
110 }
111 else
112 {
113 distinctValuesHigh |= (1L << (v - 64));
114 }
115 }
116
117 /**
118 * Increase the concurrency count by one for this slot.
119 */
120 public void updateConcurrency()
121 {
122 this.concurrentCount++;
123 }
124
125 /**
126 * Adds a value to this set, we support only positive values.
127 *
128 * @param value
129 * the value
130 * @return the scaled value
131 */
132 private int scaleIfNeeded(final int value)
133 {
134 // adjust the value according to the current scale
135 var v = value >> distinctValuesScale; // div
136
137 // make the value fit into the bit set by scaling the bit set as necessary
138 while (v >= 128)
139 {
140 // we join adjacent buckets by increasing joining the bits adjacent bits
141 // this means we lose precision, but we can still
142 var l = distinctValuesLow = BitCompression.combineAdjacentBits(distinctValuesLow);
143 l = BitCompression.compressAndShiftOddBits(l);
144
145 var h = BitCompression.combineAdjacentBits(distinctValuesHigh);
146 h = BitCompression.compressAndShiftOddBits(h);
147
148 // join them in low
149 distinctValuesLow = l | (h << 32);
150 // clear high part
151 distinctValuesHigh = 0;
152
153 // increase scale and try again
154 v = value >> (++distinctValuesScale);
155 }
156
157 return v;
158 }
159
160 /**
161 * Returns the sum of the values added to this min-max value.
162 *
163 * @return the accumulated value
164 */
165 public long getTotalValue()
166 {
167 return this.totalValue;
168 }
169
170 /**
171 * Returns the average of the values added to this min-max value.
172 *
173 * @return the average value
174 */
175 public int getAverageValue()
176 {
177 return this.count == 0 ? 0 : (int) (this.totalValue / this.count);
178
179 }
180
181 /**
182 * Returns the maximum of the values added to this min-max value.
183 *
184 * @return the maximum value
185 */
186 public int getMaximumValue()
187 {
188 return maximum == Integer.MIN_VALUE ? 0 : maximum;
189 }
190
191 /**
192 * Returns the minimum of the values added to this min-max value.
193 *
194 * @return the minimum value
195 */
196 public int getMinimumValue()
197 {
198 return minimum == Integer.MAX_VALUE ? 0 : minimum;
199 }
200
201 /**
202 * Returns the number of errors added to this min-max value.
203 */
204 public int getErrorCount()
205 {
206 return this.errorCount;
207 }
208
209 /**
210 * Returns an approximation of the distinct values added to this min-max value.
211 *
212 * @return the values
213 */
214 public double[] getValues()
215 {
216 final List<Double> values = new ArrayList<>(128);
217
218 for (int i = 0; i < 128; i++)
219 {
220 final boolean set = (i < 64) ? ((distinctValuesLow & (1L << i)) != 0)
221 : ((distinctValuesHigh & (1L << (i - 64))) != 0);
222 final double v = (1L << distinctValuesScale ) * i;
223 if (set)
224 {
225 values.add(v);
226 }
227 }
228
229 return values.stream().mapToDouble(Double::doubleValue).toArray();
230 }
231
232 /**
233 * Returns the number of values added
234 *
235 * @return the count
236 */
237 public int getCount()
238 {
239 return this.count;
240 }
241
242 /**
243 * Returns the number of concurrent measurements added
244 *
245 * @return the count
246 */
247 public int getConcurrentCount()
248 {
249 return this.concurrentCount;
250 }
251
252 /**
253 * Merges the data. Attention: This might modify the given item
254 * because we have to scale it to our scale or vice versa.
255 *
256 * @param item
257 * the other value
258 */
259 public IntTimeSeriesEntry merge(final IntTimeSeriesEntry item)
260 {
261 // we can only merge the same scale
262 if (item.distinctValuesScale > this.distinctValuesScale)
263 {
264 // we need to scale up our values
265 while (item.distinctValuesScale > this.distinctValuesScale)
266 {
267 distinctValuesLow = BitCompression.combineAdjacentBits(distinctValuesLow);
268 distinctValuesLow = BitCompression.compressAndShiftOddBits(distinctValuesLow);
269
270 distinctValuesHigh = BitCompression.combineAdjacentBits(distinctValuesHigh);
271 distinctValuesHigh = BitCompression.compressAndShiftOddBits(distinctValuesHigh);
272
273 // increase scale
274 distinctValuesScale++;
275 }
276 }
277 else if (item.distinctValuesScale < this.distinctValuesScale)
278 {
279 // we need to scale up the other values
280 while (item.distinctValuesScale < this.distinctValuesScale)
281 {
282 item.distinctValuesLow = BitCompression.combineAdjacentBits(item.distinctValuesLow);
283 item.distinctValuesLow = BitCompression.compressAndShiftOddBits(item.distinctValuesLow);
284
285 item.distinctValuesHigh = BitCompression.combineAdjacentBits(item.distinctValuesHigh);
286 item.distinctValuesHigh = BitCompression.compressAndShiftOddBits(item.distinctValuesHigh);
287
288 // increase scale
289 item.distinctValuesScale++;
290 }
291 }
292
293 maximum = Math.max(maximum, item.maximum);
294 minimum = Math.min(minimum, item.minimum);
295
296 totalValue += item.totalValue;
297 count += item.count;
298 // because we count concurrency per entry, we have to take the maximum here
299 // and cannot sum them up
300 concurrentCount = Math.max(concurrentCount, item.concurrentCount);
301 errorCount += item.errorCount;
302
303 distinctValuesLow |= item.distinctValuesLow;
304 distinctValuesHigh |= item.distinctValuesHigh;
305
306 return this;
307 }
308
309 /**
310 * {@inheritDoc}
311 */
312 @Override
313 public String toString()
314 {
315 // that is only for debugging
316 return
317 getCount() + " / " +
318 getConcurrentCount() + " / " +
319 getErrorCount() + " / " +
320 getTotalValue() + " / " +
321 getAverageValue() + " / " +
322 getMinimumValue() + " / " +
323 getMaximumValue() + " / " +
324 Arrays.toString(getValues()) + "\n";
325 }

327 @Override
328 public boolean equals(final Object obj)
329 {
330 if (this == obj)
331 {
332 return true;
333 }
334 if (obj == null)
335 {
336 return false;
337 }
338 if (getClass() != obj.getClass())
339 {
340 return false;
341 }
342 final IntTimeSeriesEntry other = (IntTimeSeriesEntry) obj;
343 if (count != other.count)
344 {
345 return false;
346 }
347 if (maximum != other.maximum)
348 {
349 return false;
350 }
351 if (minimum != other.minimum)
352 {
353 return false;
354 }
355 if (totalValue != other.totalValue)
356 {
357 return false;
358 }
359 if (errorCount != other.errorCount)
360 {
361 return false;
362 }
363 if (distinctValuesLow != other.distinctValuesLow)
364 {
365 return false;
366 }
367 if (distinctValuesHigh != other.distinctValuesHigh)
368 {
369 return false;
370 }
371 if (concurrentCount != other.concurrentCount)
372 {
373 return false;
374 }
375 if (distinctValuesScale != other.distinctValuesScale)
376 {
377 return false;
378 }
379 return true;
380 }
