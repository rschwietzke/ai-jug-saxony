21 /**
22 * Test suite for {@link LRUClockMap}. Uses plain JUnit 6 assertions only.
23 * <p>
24 * Many tests exploit the deterministic internal layout: with maxSize 4 the
25 * backing array has capacity 8 (mask 7), and small non-negative Integer keys
26 * hash to themselves, so key i lands in slot i. This makes the clock eviction
27 * order fully predictable.
28 */
29 class LRUClockMapTest

31 /**
32 * Reads the second-chance flag of a key via the package-private debug view.
33 * Fails the test if the key is not present.
34 */
35 private static <K, V> boolean secondChanceOf(final LRUClockMap<K, V> map, final K key)
36 {
37 for (final var dw : map.getDebugData())
38 {
39 if (dw != null && dw.key.equals(key))
40 {
41 return dw.secondChance;
42 }
43 }
44 throw new AssertionError("key not found in map: " + key);
45 }

707 @Nested
708 @DisplayName("Stress and invariants")
709 class StressInvariants
710 {
711 @Test
712 @DisplayName("Randomized operations keep size and value invariants")
713 void randomizedInvariantCheck()
714 {
715 final LRUClockMap<Integer, Integer> map = new LRUClockMap<>(16);
716 final Map<Integer, Integer> lastValue = new HashMap<>(); // last put value per key
717 final Random random = new Random(42L);
718
719 for (int step = 0; step < 5_000; step++)
720 {
721 final int key = random.nextInt(64);
722
723 if (random.nextInt(10) < 7)
724 {
725 map.put(key, step);
726 lastValue.put(key, step);
727 }
728 else
729 {
730 map.remove(key);
731 lastValue.remove(key);
732 }
733
734 assertTrue(map.size() <= 16, "size exceeded maxSize at step " + step);
735 assertEquals(map.size(), map.trueSize(), "size/trueSize mismatch at step " + step);
736 }
737
738 // every key reported by keys() exists and carries the last written value
739 final List<Integer> keys = map.keys();
740 assertEquals(map.size(), keys.size());
741 for (final Integer k : keys)
742 {
743 assertEquals(lastValue.get(k), map.getRaw(k), "stale value for key " + k);
744 }
745
746 // every reference key is either evicted or has the correct value,
747 // and the number of surviving reference keys matches size()
748 int survivors = 0;
749 for (final Map.Entry<Integer, Integer> e : lastValue.entrySet())
750 {
751 final Integer v = map.getRaw(e.getKey());
752 if (v != null)
753 {
754 assertEquals(e.getValue(), v, "value mismatch for key " + e.getKey());
755 survivors++;
756 }
757 }
758 assertEquals(map.size(), survivors, "map holds keys outside the reference set");
759 }
760 }
