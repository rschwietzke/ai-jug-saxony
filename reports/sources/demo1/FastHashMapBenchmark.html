26 /**
27 * Microbenchmark comparing {@link FastHashMap} (open addressing with linear
28 * probing)
29 * against standard {@link java.util.HashMap} (separate chaining with linked
30 * nodes / red-black trees).
31 *
32 * <h2>Key Microbenchmarking Design Considerations:</h2>
33 * <ul>
34 * <li><b>Hot-path overhead elimination:</b> Instead of an expensive modulo
35 * operation ({@code % size})
36 * which emits an {@code idiv} instruction (15–25 CPU cycles) and branch checks
37 * on every call,
38 * we use power-of-two sizes and bitwise masking ({@code index++ & mask}). This
39 * executes in 1 cycle
40 * and prevents arithmetic overflow issues when the counter passes
41 * {@link Integer#MAX_VALUE}.</li>
42 *
43 * <li><b>Single-threaded execution:</b> As this benchmark is strictly single-threaded,
44 * the access counter ({@code index}) is kept directly in the benchmark state without
45 * additional concurrency wrapping.</li>
46 *
47 * <li><b>Direct return instead of Blackhole:</b> For sub-10ns operations,
48 * calling {@code Blackhole.consume()}
49 * adds non-trivial method invocation overhead. Returning the value directly
50 * allows the JMH harness to
51 * sink the result using compiler intrinsics with zero overhead.</li>
52 *
53 * <li><b>Controlled pre-sizing:</b> Both maps are initialized with explicit
54 * capacities in {@link #setup()}
55 * to match the expected item count. This avoids measuring accidental load
56 * factor differences caused by
57 * different default growth thresholds (e.g. {@code FastHashMap} 0.5 vs
58 * {@code HashMap} 0.75).</li>
59 *
60 * <li><b>Clear operation semantics:</b> Accessing existing keys with
61 * {@code put} is labeled as an update
62 * ({@code updateFastMap}), since all keys were pre-populated during trial
63 * setup. It tests in-place
64 * value replacement rather than new table insertion or resizing.</li>
65 *
66 * <li><b>Note on String key hash caching:</b> {@link String#hashCode()} caches
67 * its hash internally after
68 * the first invocation. Because all keys are inserted during setup, string hash
69 * codes are already
70 * cached before measurement begins. Note that {@link FastHashMap#get} calls
71 * {@code hashCode()} repeatedly
72 * during probing, which is cheap for {@code String} but would be costly for
73 * objects without cached hashes.</li>
74 * </ul>
75 */
76 @BenchmarkMode({ Mode.Throughput, Mode.AverageTime })
77 @OutputTimeUnit(TimeUnit.NANOSECONDS)
78 @Warmup(iterations = 3, time = 1, timeUnit = TimeUnit.SECONDS)
79 @Measurement(iterations = 5, time = 1, timeUnit = TimeUnit.SECONDS)
80 @Fork(value = 2, jvmArgsAppend = { "-Xms2g", "-Xmx2g" })
81 @State(Scope.Benchmark)
82 public class FastHashMapBenchmark {
83
84 /**
85 * Powers of two allow branchless bitmasking: {@code (idx & mask)} instead of
86 * {@code (idx % size)}.
87 * This eliminates 10-25 cycles of hardware integer division latency from every
88 * benchmark iteration.
89 */
90 @Param({ "8", "128", "1024", "8192", "65536" })
91 public int size;
92
93 private FastHashMap<String, String> fastMap;
94 private Map<String, String> javaMap;
95
96 private String[] existingKeys;
97 private String[] missingKeys;
98 private String[] values;
99 private int mask;
100 private int index;
101
102 @Setup(Level.Trial)
103 public void setup() {
104 mask = size - 1;
105
106 // Pre-size both maps so initial allocation and load factors are controlled and
107 // comparable:
108 // - FastHashMap capacity is calculated for fill factor 0.5f (if supported).
109 // - HashMap capacity is calculated for default load factor 0.75f without
110 // triggering a resize.
111 fastMap = createFastMap(size);
112 javaMap = new HashMap<>((int) Math.ceil(size / 0.75f) + 1);
113
114 existingKeys = new String[size];
115 missingKeys = new String[size];
116 values = new String[size];
117
118 Random rnd = new Random(42);
119 for (int i = 0; i < size; i++) {
120 existingKeys[i] = "key_" + i + "_" + rnd.nextInt(1_000_000);
121 missingKeys[i] = "miss_" + i + "_" + rnd.nextInt(1_000_000);
122 values[i] = "val_" + i;
123
124 fastMap.put(existingKeys[i], values[i]);
125 javaMap.put(existingKeys[i], values[i]);
126 }
127 }
128
129 // ---------------------------------------------------------------------------------------------
130 // Get Hits: Key exists in map
131 // ---------------------------------------------------------------------------------------------
132
133 @Benchmark
134 public String getHitFastMap() {
135 return fastMap.get(existingKeys[index++ & mask]);
136 }
137
138 @Benchmark
139 public String getHitJavaMap() {
140 return javaMap.get(existingKeys[index++ & mask]);
141 }
142
143 // ---------------------------------------------------------------------------------------------
144 // Get Misses: Key does not exist in map (exercises probing until FREE_KEY /
145 // null node)
146 // ---------------------------------------------------------------------------------------------
147
148 @Benchmark
149 public String getMissFastMap() {
150 return fastMap.get(missingKeys[index++ & mask]);
151 }
152
153 @Benchmark
154 public String getMissJavaMap() {
155 return javaMap.get(missingKeys[index++ & mask]);
156 }
157
158 // ---------------------------------------------------------------------------------------------
159 // In-place Update: Key already exists, value is overwritten
160 // Note: This benchmarks updating existing entries, not inserting new entries
161 // into the map.
162 // ---------------------------------------------------------------------------------------------
163
164 @Benchmark
165 public String updateFastMap() {
166 int idx = index++ & mask;
167 return fastMap.put(existingKeys[idx], values[idx]);
168 }
169
170 @Benchmark
171 public String updateJavaMap() {
172 int idx = index++ & mask;
173 return javaMap.put(existingKeys[idx], values[idx]);
174 }
175
176 @SuppressWarnings("unchecked")
177 private static <K, V> FastHashMap<K, V> createFastMap(int size) {
178 try {
179 return (FastHashMap<K, V>) FastHashMap.class.getConstructor(int.class, float.class).newInstance(size, 0.5f);
180 } catch (Exception e) {
181 return new FastHashMap<>();
182 }
183 }

199 private static void fixClasspathForFork() {
200 ClassLoader cl = Thread.currentThread().getContextClassLoader();
201 if (cl instanceof java.net.URLClassLoader ucl) {
202 StringBuilder sb = new StringBuilder();
203 String currentCp = System.getProperty("java.class.path", "");
204 sb.append(currentCp);
205 for (java.net.URL url : ucl.getURLs()) {
206 if (!currentCp.contains(url.getPath())) {
207 if (sb.length() > 0) {
208 sb.append(java.io.File.pathSeparator);
209 }
210 sb.append(url.getPath());
211 }
212 }
213 System.setProperty("java.class.path", sb.toString());
214 }
