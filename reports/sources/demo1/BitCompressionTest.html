18 import org.junit.jupiter.api.DisplayName;
19 import org.junit.jupiter.api.Test;
20
21 import java.lang.reflect.Constructor;
22 import java.util.Random;
23
24 import static org.assertj.core.api.Assertions.assertThat;
25
26 @DisplayName("BitCompression Tests")
27 class BitCompressionTest
28 {
29 @Test
30 @DisplayName("Private constructor can be invoked via reflection for coverage")
31 void privateConstructor() throws Exception
32 {
33 Constructor<BitCompression> constructor = BitCompression.class.getDeclaredConstructor();
34 constructor.setAccessible(true);
35 BitCompression instance = constructor.newInstance();
36 assertThat(instance).isNotNull();
37 }
38
39 @Test
40 @DisplayName("Zero input results in zero output")
41 void zeroInput()
42 {
43 assertThat(BitCompression.combineAdjacentBits(0L)).isEqualTo(0L);
44 assertThat(BitCompression.compressAndShiftOddBits(0L)).isEqualTo(0L);
45 }
46
47 @Test
48 @DisplayName("Single bits at even and odd positions map to correct combined/compressed bits")
49 void singleBits()
50 {
51 for (int i = 0; i < 64; i++)
52 {
53 long input = 1L << i;
54 long combined = BitCompression.combineAdjacentBits(input);
55 long compressed = BitCompression.compressAndShiftOddBits(combined);
56
57 int expectedBucket = i / 2;
58 long expectedCompressed = 1L << expectedBucket;
59
60 assertThat(compressed)
61 .as("Bit position " + i + " should map to compressed bucket " + expectedBucket)
62 .isEqualTo(expectedCompressed);
63 }
64 }
65
66 @Test
67 @DisplayName("Adjacent bit pairs merge into a single compressed bit")
68 void adjacentPairs()
69 {
70 for (int pair = 0; pair < 32; pair++)
71 {
72 long bitEven = 1L << (2 * pair);
73 long bitOdd = 1L << (2 * pair + 1);
74 long pairMask = bitEven | bitOdd;
75
76 long combined = BitCompression.combineAdjacentBits(pairMask);
77 long compressed = BitCompression.compressAndShiftOddBits(combined);
78
79 assertThat(compressed)
80 .as("Pair " + pair + " should compress to bit " + pair)
81 .isEqualTo(1L << pair);
82 }
83 }
84
85 @Test
86 @DisplayName("All bits set (0xFFFFFFFFFFFFFFFFL) compresses to 0xFFFFFFFFL (32 bits set)")
87 void allBitsSet()
88 {
89 long allOnes = -1L;
90 long combined = BitCompression.combineAdjacentBits(allOnes);
91 long compressed = BitCompression.compressAndShiftOddBits(combined);
92
93 assertThat(compressed).isEqualTo(0xFFFFFFFFL);
94 assertThat(Long.bitCount(compressed)).isEqualTo(32);
95 }
96
97 @Test
98 @DisplayName("Alternating bits compress correctly")
99 void alternatingBits()
100 {
101 // 0x5555555555555555L has even bits 0, 2, 4, ..., 62 set
102 long evenBits = 0x5555555555555555L;
103 long combinedEven = BitCompression.combineAdjacentBits(evenBits);
104 long compressedEven = BitCompression.compressAndShiftOddBits(combinedEven);
105 assertThat(compressedEven).isEqualTo(0xFFFFFFFFL);
106
107 // 0xAAAAAAAAAAAAAAAA has odd bits 1, 3, 5, ..., 63 set
108 long oddBits = 0xAAAAAAAAAAAAAAAAL;
109 long combinedOdd = BitCompression.combineAdjacentBits(oddBits);
110 long compressedOdd = BitCompression.compressAndShiftOddBits(combinedOdd);
111 assertThat(compressedOdd).isEqualTo(0xFFFFFFFFL);
112 }
113
114 @Test
115 @DisplayName("Random bit patterns compress consistently with reference algorithm")
116 void randomPatterns()
117 {
118 Random rng = new Random(42);
119 for (int trial = 0; trial < 10_000; trial++)
120 {
121 long val = rng.nextLong();
122 long combined = BitCompression.combineAdjacentBits(val);
123 long compressed = BitCompression.compressAndShiftOddBits(combined);
124
125 long expected = 0;
126 for (int k = 0; k < 32; k++)
127 {
128 boolean bit2k = (val & (1L << (2 * k))) != 0;
129 boolean bit2k1 = (val & (1L << (2 * k + 1))) != 0;
130 if (bit2k || bit2k1)
131 {
132 expected |= (1L << k);
133 }
134 }
135
136 assertThat(compressed)
137 .as("Mismatch on random value: 0x" + Long.toHexString(val))
138 .isEqualTo(expected);
139 }
140 }
