406 private enum Op
407 {
408 AND, OR, ANDNOT, XOR
409 }
410
411 private static long[] randomArray(final Random random, final int length)
412 {
413 final long[] a = new long[length];
414 for (int i = 0; i < length; i++)
415 {
416 a[i] = random.nextLong();
417 }
418 return a;
419 }
420
421 private static long sumBitCount(final long[] a, final int offset, final int numWords)
422 {
423 long total = 0;
424 for (int i = offset; i < offset + numWords; i++)
425 {
426 total += Long.bitCount(a[i]);
427 }
428 return total;
429 }
430
431 private static long sumBitCountOp(final long[] a, final long[] b, final int offset, final int numWords, final Op op)
432 {
433 long total = 0;
434 for (int i = offset; i < offset + numWords; i++)
435 {
436 final long combined = switch (op)
437 {
438 case AND -> a[i] & b[i];
439 case OR -> a[i] | b[i];
440 case ANDNOT -> a[i] & ~b[i];
441 case XOR -> a[i] ^ b[i];
442 };
443 total += Long.bitCount(combined);
444 }
445 return total;
446 }
