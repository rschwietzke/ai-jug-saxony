18 import static org.junit.jupiter.api.Assertions.assertEquals;
19
20 import java.util.Random;

22 import org.junit.jupiter.api.Test;

24 class BitCompressionTest
25 {
26 @Test
27 void combineAdjacentBitsMapsEveryInputBit()
28 {
29 for (int bitIndex = 0; bitIndex < Long.SIZE; bitIndex++)
30 {
31 final long input = 1L << bitIndex;
32 final long expected = input | (bitIndex == Long.SIZE - 1 ? 0 : 1L << (bitIndex + 1));
33
34 assertEquals(expected, BitCompression.combineAdjacentBits(input), "bit " + bitIndex);
35 }
36 }
37
38 @Test
39 void compressAndShiftOddBitsMapsOddBitsAndIgnoresEvenBits()
40 {
41 for (int outputBit = 0; outputBit < Integer.SIZE; outputBit++)
42 {
43 assertEquals(0, BitCompression.compressAndShiftOddBits(1L << (outputBit * 2)),
44 "even input bit " + (outputBit * 2));
45 assertEquals(1L << outputBit,
46 BitCompression.compressAndShiftOddBits(1L << (outputBit * 2 + 1)),
47 "odd input bit " + (outputBit * 2 + 1));
48 }
49 }
50
51 @Test
52 void compressionReturnsAZeroExtendedThirtyTwoBitValue()
53 {
54 assertEquals(0, BitCompression.compressAndShiftOddBits(0));
55 assertEquals(0, BitCompression.compressAndShiftOddBits(0x5555555555555555L));
56 assertEquals(0xFFFFFFFFL, BitCompression.compressAndShiftOddBits(0xAAAAAAAAAAAAAAAAL));
57 assertEquals(0xFFFFFFFFL, BitCompression.compressAndShiftOddBits(-1L));
58 assertEquals(0x80000000L, BitCompression.compressAndShiftOddBits(Long.MIN_VALUE));
59 }
60
61 @Test
62 void operationsMatchIndependentBitByBitReferences()
63 {
64 final Random random = new Random(0xB17C0FFEE0L);
65
66 for (int iteration = 0; iteration < 10_000; iteration++)
67 {
68 final long input = random.nextLong();
69 final long combined = BitCompression.combineAdjacentBits(input);
70 final long compressed = BitCompression.compressAndShiftOddBits(input);
71 final long pairOccupancy = BitCompression.compressAndShiftOddBits(combined);
72
73 assertEquals(combineReference(input), combined, "combine iteration " + iteration);
74 assertEquals(compressOddBitsReference(input), compressed, "compress iteration " + iteration);
75 assertEquals(pairOccupancyReference(input), pairOccupancy, "composition iteration " + iteration);
76 assertEquals(0, compressed >>> Integer.SIZE, "upper bits at iteration " + iteration);
77 }
78 }

80 private static long combineReference(final long input)
81 {
82 long result = 0;
83 for (int bitIndex = 0; bitIndex < Long.SIZE; bitIndex++)
84 {
85 if ((input & (1L << bitIndex)) != 0)
86 {
87 result |= 1L << bitIndex;
88 if (bitIndex + 1 < Long.SIZE)
89 {
90 result |= 1L << (bitIndex + 1);
91 }
92 }
93 }
94 return result;
95 }
96
97 private static long compressOddBitsReference(final long input)
98 {
99 long result = 0;
100 for (int outputBit = 0; outputBit < Integer.SIZE; outputBit++)
101 {
102 if ((input & (1L << (outputBit * 2 + 1))) != 0)
103 {
104 result |= 1L << outputBit;
105 }
106 }
107 return result;
108 }
109
110 private static long pairOccupancyReference(final long input)
111 {
112 long result = 0;
113 for (int pair = 0; pair < Integer.SIZE; pair++)
114 {
115 if (((input >>> (pair * 2)) & 0b11L) != 0)
116 {
117 result |= 1L << pair;
118 }
119 }
120 return result;
121 }
