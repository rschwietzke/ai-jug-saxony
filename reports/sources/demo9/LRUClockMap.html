24 /**
25 * This is a simple LRU map implementation based on the clock algorithm.
26 * Not thread-safe! Null key and value support was removed.
27 * The LRU part is not really predictable, but it is fast and simple.
28 * We don't keep a linked list, rather a marker that give every entry a second chance.
29 * If the second chance is false, we can evict this entry. This can happen
30 * rather early. In general, it is still a good enough LRU implementation.
31 *
32 * This makes it memory and CPU cache efficient.
33 *
34 * https://www.geeksforgeeks.org/operating-systems/second-chance-or-clock-page-replacement-policy/
35 *
36 * @since 9.1.0
37 * @author René Schwietzke (Xceptance Software Technologies GmbH)
38 */
39 public class LRUClockMap<K, V>
40 {
41 /**
42 * Keys and values
43 */
44 private Wrapper<K, V>[] data;
45
46 /**
47 * Current cache size
48 */
49 private int size;
50
51 /**
52 * Max cache size
53 */
54 private final int maxSize;
55
56 /**
57 * Mask to calculate the original position
58 * */
59 private int mask;
60
61 /**
62 * Our clock hand, the next position in the array to check for eviction
63 */
64 private int clockHand = 0;
65
66 /**
67 * Wrapper class for key, value, and secondChance bit.
68 * When secondChance is true we skip it when the clock
69 * is checked. If it is false, we dispose the entry.
70 */
71 private static class Wrapper<K, V>
72 {
73 final K key;
74 V value;
75 boolean secondChance;
76
77 /**
78 * Create a new wrapper entry
79 * @param key the key
80 * @param value the value
81 */
82 Wrapper(final K key, final V value)
83 {
84 this.key = key;
85 this.value = value;
86 // don't give up on a new entry so easily
87 this.secondChance = true;
88 }
89
90 /**
91 * For debugging purposes
92 */
93 @Override
94 public String toString()
95 {
96 return "[" + key.toString() + ", " + value.toString() + ", " + secondChance + "]";
97 }
98 }

100 /**
101 * Creates a new LRU map with the given maximum size.
102 *
103 * @param maxSize the maximum number of entries to hold
104 */
105 @SuppressWarnings("unchecked")
106 public LRUClockMap(final int maxSize)
107 {
108 if (maxSize < 4)
109 {
110 throw new IllegalArgumentException( "MaxSize must be at least 4" );
111 }
112 this.maxSize = maxSize;
113
114 final int capacity = arraySize(maxSize, 0.50f);
115 this.mask = capacity - 1;
116
117 this.data = (Wrapper<K, V>[]) new Wrapper[capacity];
118 }
119
120 /**
121 * Mix the hash to have a better distribution.
122 * Inspired by OpenJDK HashMap.
123 *
124 * @param h the original hash
125 * @return the mixed hash
126 */
127 private int mixHash(final int h)
128 {
129 return h ^ (h >>> 16);
130 }
131
132 /**
133 * Read data without touching the clock flag. This is mostly for testing purposes
134 * or in case you really don't want to affect the LRU state.
135 *
136 * @param key what key to look for
137 * @return the value for the key or null if not founds
138 */
139 public V getRaw(final K key)
140 {
141 Objects.requireNonNull(key, "Key must not be null");
142 int ptr = mixHash(key.hashCode()) & this.mask;
143
144 while (true)
145 {
146 final Wrapper<K, V> w = this.data[ptr];
147
148 if (w == null)
149 {
150 return null;
151 }
152 else if (w.key.equals(key))
153 {
154 return w.value;
155 }
156
157 ptr = (ptr + 1) & this.mask;
158 }
159 }
160
161 /**
162 * Get a value from the map. This will set the second chance flag
163 * if the entry is found.
164 *
165 * @param key what key to look for
166 * @return the value for the key or null if not found
167 */
168 public V get(final K key)
169 {
170 Objects.requireNonNull(key, "Key must not be null");
171 int ptr = mixHash(key.hashCode()) & this.mask;
172 final Wrapper<K, V> w = this.data[ptr];
173
174 if (w == null)
175 {
176 return null;
177 }
178
179 if (w.key.equals(key))
180 {
181 if (!w.secondChance)
182 {
183 // we have to give it a second chance
184 // but only if it is not already set
185 // saves a write operation
186 w.secondChance = true;
187 }
188 return w.value;
189 }
190 else
191 {
192 // we have to do linear probing now
193 // this is the slow path and hence its own method
194 // to keep the main get method as lean as possible
195 // for better inlining
196 return expensiveGet(key, ptr);
197 }
198 }
199
200 /**
201 * We have this code here to make the normal get as lean as possible
202 * for better inlining. This is the slow path.
203 *
204 * @param key what key to look for
205 * @param ptr the current position in the array
206 * @return the value for the key or null if not found
207 */
208 private V expensiveGet(final K key, int ptr)
209 {
210 // we start with the next position because we already checked ptr
211 // and missed there
212 while (true)
213 {
214 ptr = (ptr + 1) & this.mask;
215 final Wrapper<K, V> w = this.data[ptr];
216
217 if (w == null)
218 {
219 return null;
220 }
221
222 if (w.key.equals(key))
223 {
224 if (!w.secondChance)
225 {
226 // we have to give it a second chance
227 // but only if it is not already set
228 // saves a write operation
229 w.secondChance = true;
230 }
231 return w.value;
232 }
233
234 // not found, continue searching
235 }
236 }

238 /**
239 * Associates the specified value with the specified key in this map.
240 * If the map previously contained a mapping for the key, the old
241 * value is replaced. Starts eviction if the max size is reached.
242 *
243 * @param key key with which the specified value is to be associated
244 * @param value value to be associated with the specified key
245 * @return the previous value associated with key, or null if there was no mapping for key.
246 */
247 public V put(final K key, final V value)
248 {
249 Objects.requireNonNull(key, "Key must not be null");
250 Objects.requireNonNull(value, "Value must not be null");
251 final int ptr = mixHash(key.hashCode()) & this.mask;
252
253 // evict before inserting a new entry helps to avoid
254 // pushing it out again immediately
255 if (this.size == maxSize)
256 {
257 // if this is going to be an insert, we have to evict
258 // otherwise just update
259 final V oldValue = update(key, value, ptr);
260 if (oldValue != null)
261 {
262 // we just updated an existing entry, no eviction needed
263 return oldValue;
264 }
265 // ok, we have not updated, so we have to kick one out
266 evict();
267 }
268
269 // put the entry, might replace an existing one or be added, we have enough space
270 // at this point for sure
271 return putInternal(key, value, ptr);
272 }
273
274 /**
275 * Internal put method that does not evict
276 *
277 * @param key the key to put
278 * @param value the value to put
279 * @param ptr the position to start searching
280 *
281 * @return the old value or null if there was none
282 */
283 private V putInternal(final K key, final V value, int ptr)
284 {
285 while (true)
286 {
287 final Wrapper<K, V> current = this.data[ptr];
288
289 if (current == null)
290 {
291 // found a free slot
292 this.data[ptr] = new Wrapper<>(key, value);
293 // we added one entry, so size is increased
294 this.size++;
295
296 return null;
297 }
298 else if (current.key.equals(key))
299 {
300 // key already exists, replace value
301 final V oldValue = current.value;
302 current.value = value;
303 current.secondChance = true;
304
305 return oldValue;
306 }
307
308 ptr = (ptr + 1) & this.mask; // move to next position
309 }
310 }
311
312 /**
313 * Internal method to combine get and put functionality but
314 * without insertion because we might have to evict first.
315 *
316 * @param key the key to put
317 * @param value the value to put
318 * @param ptr the position to start searching
319 *
320 * @return the old value or null if there was none
321 */
322 private V update(final K key, final V value, int ptr)
323 {
324 while (true)
325 {
326 final Wrapper<K, V> current = this.data[ptr];
327
328 if (current == null)
329 {
330 // no entry found, we don't do anything here
331 // just tell
332 return null;
333 }
334 else if (current.key.equals(key))
335 {
336 // key already exists, replace value
337 final V oldValue = current.value;
338 current.value = value;
339 current.secondChance = true;
340
341 return oldValue;
342 }
343
344 ptr = (ptr + 1) & this.mask; // move to next position
345 }
346 }
347
348 /**
349 * Removes the mapping for a key from this map if it is present.
350 *
351 * @param key key whose mapping is to be removed from the map
352 * @return the previous value associated with key, or null if there was no mapping for key.
353 */
354 public V remove(final K key)
355 {
356 Objects.requireNonNull(key, "Key must not be null");
357 int ptr = mixHash(key.hashCode()) & this.mask;
358
359 while (true)
360 {
361 final Wrapper<K, V> w = this.data[ptr];
362
363 if (w == null)
364 {
365 // Key not found
366 return null;
367 }
368
369 if (w.key.equals(key))
370 {
371 final V oldValue = w.value;
372 // free this entry and fix array up
373 freePositionAndAdjustArray(ptr);
374
375 return oldValue;
376 }
377
378 ptr = (ptr + 1) & this.mask;
379 }
380 }
381
382 /**
383 * Looks for an entry that has not been touched recently and can be evicted.
384 * This is not truly LRU but according to common knowledge, close enough and
385 * very cheap.
386 */
387 private void evict()
388 {
389 while (true)
390 {
391 final Wrapper<K, V> w = this.data[clockHand];
392
393 if (w != null)
394 {
395 if (w.secondChance == false)
396 {
397 // free this entry and fix array up
398 freePositionAndAdjustArray(clockHand);
399 return;
400 }
401 else
402 {
403 // gave it a second chance, now this is gone
404 w.secondChance = false;
405 }
406 }
407
408 clockHand = (clockHand + 1) & this.mask; // move to next position
409 }
410 }
411
412 /**
413 * We remove this position from the map. Because we are a linear map, we have
414 * to shift all entries after this position one step to the left until we
415 * hit a free position to ensure we don't break the linear chaining.
416 *
417 * @param ptr the position to remove
418 */
419 private void freePositionAndAdjustArray(final int ptr)
420 {
421 this.data[ptr] = null;
422 this.size--; // we removed one entry, so size is reduced
423
424 // Shift all entries after this position if needed
425 int currentPtr = ptr;
426 while (true)
427 {
428 currentPtr = (currentPtr + 1) & this.mask;
429
430 final Wrapper<K, V> w = this.data[currentPtr];
431 if (w == null)
432 {
433 break; // no more entries to shift
434 }
435
436 this.data[currentPtr] = null; // free this entry
437 realign(w); // put it back to the map, it will find a new position eventually
438 }
439 }
440
441 /**
442 * Internal reput to account for removed entries. We do not have
443 * double keys in the map, so we can safely ignore that fact.
444 *
445 * @param entry the wrapper to put
446 */
447 private void realign(final Wrapper<K, V> entry)
448 {
449 int ptr = mixHash(entry.key.hashCode()) & this.mask;
450
451 while (true)
452 {
453 final Wrapper<K, V> current = this.data[ptr];
454
455 if (current == null)
456 {
457 // found a free slot
458 this.data[ptr] = entry;
459 return;
460 }
461 ptr = (ptr + 1) & this.mask; // move to next position
462 }
463 }

484 /**
485 * Returns how many slots are really occupied, should always match
486 * size but for testing, we expose that.
487 *
488 * @return the occupied slot count
489 */
490 public int trueSize()
491 {
492 int count = 0;
493 for (var w : this.data)
494 {
495 if (w != null)
496 {
497 count++;
498 }
499 }
500 return count;
501 }

503 /**
504 * Returns a list of all keys. This is mainly for testing purposes.
505 * This does not cause LRU effects. This is in order of the internal array,
506 * but without the free slots.
507 *
508 * @return a list of keys
509 */
510 public List<K> keys()
511 {
512 final List<K> result = new ArrayList<>();
513
514 final int length = this.data.length;
515 for (int i = 0; i < length; i++)
516 {
517 final Wrapper<K, V> w = this.data[i];
518 if (w != null)
519 {
520 result.add(w.key);
521 }
522 }
523
524 return result;
525 }

536 /**
537 * Return the least power of two greater than or equal to the specified value.
538 *
539 * <p>Note that this function will return 1 when the argument is 0.
540 *
541 * @param x a long integer smaller than or equal to 2<sup>62</sup>.
542 * @return the least power of two greater than or equal to the specified value.
543 */
544 private static long nextPowerOfTwo(long x)
545 {
546 if ( x == 0 )
547 {
548 return 1;
549 }
550 x--;
551 x |= x >> 1;
552 x |= x >> 2;
553 x |= x >> 4;
554 x |= x >> 8;
555 x |= x >> 16;
556
557 return (x | x >> 32) + 1;
558 }
559
560 /** Returns the least power of two smaller than or equal to 2<sup>30</sup> and larger than or equal to <code>Math.ceil( expected / f )</code>.
561 *
562 * @param expected the expected number of elements in a hash table.
563 * @param f the load factor.
564 * @return the minimum possible size for a backing array.
565 * @throws IllegalArgumentException if the necessary size is larger than 2<sup>30</sup>.
566 */
567 private static int arraySize( final int expected, final float f )
568 {
569 final long s = Math.max(2, nextPowerOfTwo((long)Math.ceil(expected / f)));
570 if (s > (1 << 30))
571 {
572 throw new IllegalArgumentException("Too large (" + expected + " expected elements with load factor " + f + ")");
573 }
574 return (int)s;
575 }

577 @Override
578 public String toString()
579 {
580 var sj = new StringJoiner(",\n", "LRUClockMap{\n", "\n}");
581
582 final int length = Math.min(this.data.length, 1024);
583 for (int i = 0; i < length; i++)
584 {
585 final Wrapper<K, V> w = this.data[i];
586 if (w == null)
587 {
588 sj.add(i + " FREE");
589 }
590 else
591 {
592 sj.add(i + " " + new DebugWrapper<K, V>(w, i).toString());
593 }
594 }
595 sj.add("clockHand: " + clockHand);
596 sj.add("size: " + size);
597 sj.add("maxSize: " + maxSize);
598
599 return sj.toString();
600 }

602 class DebugWrapper<DK, DV>
603 {
604 public final DK key;
605 public final DV value;
606 public final boolean secondChance;
607 public final int currentPosition;
608 public final int truePosition;
609
610 DebugWrapper(Wrapper<DK, DV> wrapper, int currentPosition)
611 {
612 this.key = wrapper.key;
613 this.value = wrapper.value;
614 this.secondChance = wrapper.secondChance;
615 this.currentPosition = currentPosition; // will be set later
616 this.truePosition = mixHash(wrapper.key.hashCode()) & mask; // calculate the true position
617 }
618
619 @Override
620 public String toString()
621 {
622 return "[" + key.toString() + ", " + value.toString() + ", " + mixHash(key.hashCode()) + ", " + secondChance + ", " + truePosition + "]";
623 }
624 }
625
626 List<DebugWrapper<K, V>> getDebugData()
627 {
628 final var result = new ArrayList<DebugWrapper<K, V>>(this.data.length);
629 for (int i = 0; i < this.data.length; i++)
630 {
631 final Wrapper<K, V> w = this.data[i];
632 if (w == null)
633 {
634 result.add(null);
635 }
636 else
637 {
638 result.add(new DebugWrapper<>(w, i));
639 }
640 }
641
642 return result;
643 }
