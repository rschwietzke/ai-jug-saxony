16 package com.xceptance.xlt.report.util;
17
18 import java.util.Arrays;
19
20 import com.xceptance.xlt.report.util.lucene.BitUtil;
21
22 /**
23 * The {@link RuntimeHistogram} class calculates any percentile from the <code>int</code> values added. In contrast to
24 * other implementations, this class does not store any value added, but counts the occurrences of each value. This
25 * approach saves memory if the values added are in roughly the same range.
26 */
27 public class RuntimeHistogram
28 {
29 /**
30 * The default precision, power of 2 for shifting aka this is 1<<0 = 1 (no precision loss).
31 */
32 private static final int DEFAULT_PRECISION = 1;
33
34 /**
35 * The buckets allocated so far.
36 */
37 private int[] countPerBucket;
38
39 /**
40 * Value at the first index.
41 */
42 private int firstIndexValue = Integer.MAX_VALUE;
43
44 /**
45 * Value at the last index.
46 */
47 private int lastIndexValue = Integer.MIN_VALUE;
48
49 /**
50 * What is the desired precision? We accept loss of precision in order
51 * to save memory.
52 */
53 private final int precision;
54
55 /**
56 * The number of values added so far
57 */
58 private int valueCount;
59
60 /**
61 * Constructor.
62 */
63 public RuntimeHistogram()
64 {
65 this(DEFAULT_PRECISION);
66 }
67
68 /**
69 * Constructor. Precision defines the size of each bucket. E.g. a precision of 8 means that values 0-7 go into
70 * that bucket. To be fast, this must be a power of 2.
71 *
72 * @param precision
73 * the precision to use
74 */
75 public RuntimeHistogram(final int precision)
76 {
77 // get us the power of two for shifting
78 final var nPoT = BitUtil.nextHighestPowerOfTwo(precision);
79 this.precision = Integer.numberOfTrailingZeros(nPoT);
80 }
81
82 /**
83 * Adds a value to this median calculator.
84 *
85 * @param value
86 * the value to add
87 */
88 public void addValue(final int value)
89 {
90 final int index = value >> this.precision;
91 this.valueCount++;
92
93 // grow/shift values array if necessary
94 if (index < this.firstIndexValue)
95 {
96 grow(index);
97 }
98 else if (index > this.lastIndexValue)
99 {
100 grow(index);
101 }
102 else
103 {
104 this.countPerBucket[index - this.firstIndexValue]++;
105 }
106 }
107
108 /**
109 * Grows the bucket array by the specified number of buckets.
110 *
111 * @param the new index position to cover
112 */
113 private void grow(final int newIndexPositionToSupport)
114 {
115 // ok, to avoid frequent checks for the valueCount == 0 we check here
116 // for out of bound values when we do that for the first time
117 if (firstIndexValue == Integer.MAX_VALUE)
118 {
119 this.countPerBucket = new int[1];
120 this.firstIndexValue = this.lastIndexValue = newIndexPositionToSupport;
121 this.countPerBucket[0] = 1;
122
123 return;
124 }
125
126 if (newIndexPositionToSupport < this.firstIndexValue)
127 {
128 // grow left and shift our contents to the right
129 final int delta = this.firstIndexValue - newIndexPositionToSupport;
130 final int[] oldCountPerBucket = this.countPerBucket;
131
132 this.countPerBucket = new int[this.countPerBucket.length + delta];
133 // copy and shift existing values to the right
134 System.arraycopy(oldCountPerBucket, 0, this.countPerBucket, delta, oldCountPerBucket.length);
135
136 this.firstIndexValue = newIndexPositionToSupport;
137 this.countPerBucket[0] = 1;
138 }
139 else
140 {
141 // just increase the size
142 final int delta = newIndexPositionToSupport - this.lastIndexValue;
143 this.countPerBucket = Arrays.copyOf(this.countPerBucket, this.countPerBucket.length + delta);
144
145 this.lastIndexValue = newIndexPositionToSupport;
146 this.countPerBucket[this.countPerBucket.length - 1] = 1;
147
148 }
149 }
150
151 /**
152 * Returns the number of values added.
153 *
154 * @return the number of values added
155 */
156 public int getValueCount()
157 {
158 return valueCount;
159 }
160
161 /**
162 * Returns the median of the values added.
163 *
164 * @return the median value
165 */
166 public double getMedianValue()
167 {
168 return getPercentile(50.0);
169 }
170
171 /**
172 * Returns the p-th quantile of the values added.
173 *
174 * @param p
175 * the p 0.0 &lt; p &lt;= 1.0
176 * @return the p-th percentile
177 */
178 public double getQuantile(final double p)
179 {
180 return getPercentile(p * 100.0);
181 }
182
183 /**
184 * Returns the p-th percentile of the values added.
185 *
186 * @param p
187 * the p (0 &lt;= p &le; 100)
188 * @return the p-th percentile
189 */
190 public double getPercentile(final double p)
191 {
192 // we don't support 0 but for comparison with other algorithms
193 // we need that
194 if (p < 0.0 || p > 100.0)
195 {
196 throw new IllegalArgumentException("Value of parameter 'p' must be in range (0, 100], but was " + p);
197 }
198
199 // see https://de.wikipedia.org/wiki/Quantil#Berechnung_empirischer_Quantile
200
201 double value;
202
203 if (valueCount == 0)
204 {
205 value = 0.0;
206 }
207 else if (p == 0.0)
208 {
209 value = firstIndexValue << precision;
210 }
211 else if (p == 100.0)
212 {
213 value = lastIndexValue << precision;
214 }
215 else
216 {
217 final double np = ((double)valueCount) * (p / 100.0d);
218
219 // even number of values -> mean of two adjacent values
220 if ((np % 1.0) == 0.0)
221 {
222 // get two adjacent values and calculate the mean of both
223 final int value1 = getValueByCount(np);
224 final int value2 = getValueByCount(np + 1);
225
226 value = (value1 + value2) / 2.0d;
227 }
228 else
229 {
230 // just get the corresponding value when odd
231 value = getValueByCount(Math.ceil(np));
232 }
233 }
234
235 return value;
236 }
237
238 /**
239 * Returns if the histogram is empty.
240 *
241 * @return true if empty, false otherwise
242 */
243 public boolean isEmpty()
244 {
245 return valueCount == 0;
246 }
247
248 /**
249 * Returns the number of values that fall into the specified value range.
250 *
251 * @param start the start value (inclusive)
252 * @param end the end value (inclusive)
253 *
254 * @return the number of values in the range
255 */
256 public long getCountForValue(final int start, final int end)
257 {
258 if (start > end)
259 {
260 throw new IllegalArgumentException("Start value must be less than or equal to end value");
261 }
262
263 if (valueCount == 0)
264 {
265 // there is nothing stored
266 return 0;
267 }
268
269 // Convert values to bucket indices
270 final int startIndex = start >> this.precision;
271 final int endIndex = end >> this.precision;
272
273 // Check if range is completely outside our data
274 if (endIndex < this.firstIndexValue || startIndex > this.lastIndexValue)
275 {
276 return 0;
277 }
278
279 // Clamp indices to our actual bucket range and move the index
280 // by our minimal index value offset, in case it is not zero, we
281 // safe us the empty ranges at the begin
282 final int firstBucket = Math.max(0, startIndex - this.firstIndexValue);
283 final int lastBucket = Math.min(this.countPerBucket.length - 1, endIndex - this.firstIndexValue);
284
285 // Sum up counts in the relevant buckets
286 long count = 0;
287 for (int i = firstBucket; i <= lastBucket; i++)
288 {
289 count += this.countPerBucket[i];
290 }
291
292 return count;
293 }
294
295 /**
296 * Returns the value that belong to the last index entry which
297 * fulfilled the condition of having at least the given number of values.
298 *
299 * @param totalCount the total count to reach
300 * @return the value corrected to precision
301 */
302 private int getValueByCount(final double totalCount)
303 {
304 // find the bucket that holds the value with the given index
305 int bucketIndex = -1;
306 double count = 0;
307
308 while (count < totalCount)
309 {
310 count += countPerBucket[++bucketIndex];
311 }
312
313 // reconstruct the value
314 return (firstIndexValue + bucketIndex) << precision;
315 }
316
317 /**
318 * Returns the number of allocated buckets.
319 *
320 * @return the number of buckets used
321 */
322 public int getNumberOfBuckets()
323 {
324 return countPerBucket != null ? countPerBucket.length : 0;
325 }
326
327 /**
328 * Returns the precision used. 1 means, we keep all values as is.
329 * 2 means, we reduce the result precision by a factor of 2, 4 means by a factor of 4, etc.
330 *
331 * @return the precision
332 */
333 public int getPrecision()
334 {
335 return 1 << precision;
336 }
