61 @Test
62 public void growLeftAndRight()
63 {
64 RuntimeHistogram hist = new RuntimeHistogram(1);
65 hist.addValue(50);
66 assertEquals(1, hist.getNumberOfBuckets());
67
68 // Grow right
69 hist.addValue(55);
70 assertEquals(6, hist.getNumberOfBuckets()); // 50 to 55 inclusive is 6 buckets
71
72 // Grow left
73 hist.addValue(45);
74 assertEquals(11, hist.getNumberOfBuckets()); // 45 to 55 inclusive is 11 buckets
75
76 // Add within existing range
77 hist.addValue(48);
78 assertEquals(11, hist.getNumberOfBuckets());
79 assertEquals(4, hist.getValueCount());
80 }
81
82 @Test
83 public void percentilesWithEvenAndOddCounts()
84 {
85 RuntimeHistogram hist = new RuntimeHistogram(1);
86 hist.addValue(10);
87 hist.addValue(20);
88 hist.addValue(30);
89
90 // Odd count = 3
91 // Median is at 50%: np = 3 * 0.5 = 1.5 -> ceil(1.5) = 2 -> second value = 20
92 assertEquals(20.0, hist.getMedianValue());
93 assertEquals(10.0, hist.getPercentile(0.0));
94 assertEquals(30.0, hist.getPercentile(100.0));
95
96 // Add 4th value: 40 -> even count = 4
97 // 50% -> np = 4 * 0.5 = 2.0 (integer) -> mean of value 2 (20) and value 3 (30) = 25.0
98 hist.addValue(40);
99 assertEquals(25.0, hist.getMedianValue());
100 }

112 @Test
113 public void getCountForValue()
114 {
115 RuntimeHistogram hist = new RuntimeHistogram(1);
116 assertEquals(0, hist.getCountForValue(0, 100)); // Empty
117
118 hist.addValue(10);
119 hist.addValue(20);
120 hist.addValue(20);
121 hist.addValue(30);
122
123 assertEquals(1, hist.getCountForValue(10, 10));
124 assertEquals(2, hist.getCountForValue(20, 20));
125 assertEquals(3, hist.getCountForValue(10, 20));
126 assertEquals(4, hist.getCountForValue(0, 100));
127 assertEquals(4, hist.getCountForValue(10, 30));
128
129 // Outside ranges
130 assertEquals(0, hist.getCountForValue(0, 5));
131 assertEquals(0, hist.getCountForValue(35, 100));
132
133 // Partially overlapping ranges
134 assertEquals(1, hist.getCountForValue(5, 15));
135 assertEquals(1, hist.getCountForValue(25, 35));
136
137 // Invalid range
138 assertThrows(IllegalArgumentException.class, () -> hist.getCountForValue(50, 40));
139 }

141 @Test
142 public void withPrecisionScaling()
143 {
144 // Precision 8 -> values 0-7 bucket 0, 8-15 bucket 1, etc.
145 RuntimeHistogram hist = new RuntimeHistogram(8);
146 hist.addValue(5); // index 0 -> reconstructed value = 0 << 3 = 0
147 hist.addValue(12); // index 1 -> reconstructed value = 1 << 3 = 8
148 hist.addValue(20); // index 2 -> reconstructed value = 2 << 3 = 16
149
150 assertEquals(3, hist.getValueCount());
151 assertEquals(3, hist.getNumberOfBuckets());
152 assertEquals(8.0, hist.getMedianValue());
153 assertEquals(0.0, hist.getPercentile(0.0));
154 assertEquals(16.0, hist.getPercentile(100.0));
155 }
