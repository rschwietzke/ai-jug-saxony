18 import static org.junit.jupiter.api.Assertions.assertEquals;
19 import static org.junit.jupiter.api.Assertions.assertTrue;
20
21 import java.util.Random;
22
23 import org.junit.jupiter.api.Test;
24
25 /**
26 * Tests for the {@link BitCompression} helper, the two step primitive that
27 * {@code IntTimeSeriesEntry} uses to halve the resolution of its approximate
28 * distinct value bitmap.
29 */
30 public class BitCompressionTest
31 {
32 @Test
33 public void combineAdjacentBitsBasics()
34 {
35 assertEquals(0L, BitCompression.combineAdjacentBits(0L));
36 assertEquals(-1L, BitCompression.combineAdjacentBits(-1L));
37 // 0b101 -> 0b101 | 0b1010 = 0b1111
38 assertEquals(0b1111L, BitCompression.combineAdjacentBits(0b101L));
39 // every bit is ORed with its right neighbour -> never fewer bits set
40 assertTrue(Long.bitCount(BitCompression.combineAdjacentBits(0x8000_0000_0000_0001L)) >= 2);
41 }
42
43 @Test
44 public void compressAndShiftOddBitsBasics()
45 {
46 assertEquals(0L, BitCompression.compressAndShiftOddBits(0L));
47 // only odd bits (positions 1, 3, 5, ...) are extracted:
48 // 0xAAAA... has all odd bits set -> 32 ones packed into the lower 32 bits
49 assertEquals(0xFFFF_FFFFL, BitCompression.compressAndShiftOddBits(0xAAAA_AAAA_AAAA_AAAAL));
50 // 0x5555... has all even bits set -> nothing to extract
51 assertEquals(0L, BitCompression.compressAndShiftOddBits(0x5555_5555_5555_5555L));
52 // result never uses more than the lower 32 bits
53 assertTrue(BitCompression.compressAndShiftOddBits(-1L) <= 0xFFFF_FFFFL);
54 }
55
56 @Test
57 public void singleBitMapsToHalfItsIndex()
58 {
59 // one set bit at position j in the old word must end up at position j >> 1
60 for (int j = 0; j < 64; j++)
61 {
62 final long combined = BitCompression.combineAdjacentBits(1L << j);
63 final long compressed = BitCompression.compressAndShiftOddBits(combined);
64 final long expected = 1L << (j >> 1);
65 assertEquals(expected, compressed, "mapping of bit " + j);
66 }
67 }
68
69 @Test
70 public void adjacentPairsCollapseIntoOneBit()
71 {
72 // if both members of a pair (2k, 2k+1) are set, the result is still a single bit at k
73 for (int k = 0; k < 32; k++)
74 {
75 final long pair = 3L << (2 * k);
76 final long compressed = BitCompression.compressAndShiftOddBits(BitCompression.combineAdjacentBits(pair));
77 assertEquals(1L, Long.bitCount(compressed), "expected exactly one bit for pair " + k);
78 assertEquals(1L << k, compressed, "expected bit at position " + k);
79 }
80 }
81
82 @Test
83 public void compressionOresAdjacentBuckets()
84 {
85 // The combined operation must be semantically: new bit k is set iff at least one of
86 // the old bits 2k or 2k+1 was set.
87 final Random random = new Random(42);
88 for (int trial = 0; trial < 500; trial++)
89 {
90 final long mask = random.nextLong();
91 long expected = 0;
92 for (int k = 0; k < 32; k++)
93 {
94 if ((mask & (3L << (2 * k))) != 0)
95 {
96 expected |= (1L << k);
97 }
98 }
99 final long actual = BitCompression.compressAndShiftOddBits(BitCompression.combineAdjacentBits(mask));
100 assertEquals(expected, actual, "mask " + Long.toHexString(mask));
101 }
102 }
103
104 @Test
105 public void repeatedCompressionHalvesPositionsAgain()
106 {
107 // one fold moves bit j to j >> 1; a second fold of the *same* word must move it to j >> 2
108 for (int j = 2; j < 64; j++)
109 {
110 final long first = BitCompression.compressAndShiftOddBits(BitCompression.combineAdjacentBits(1L << j));
111 final long second = BitCompression.compressAndShiftOddBits(BitCompression.combineAdjacentBits(first));
112 assertEquals(1L << (j >> 2), second, "double fold of bit " + j);
113 }
114 }
