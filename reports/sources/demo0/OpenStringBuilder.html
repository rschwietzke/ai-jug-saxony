118 /**
119 * Updates the length of the builder by either dropping the last characters or adding filler of Unicode zero.
120 *
121 * @param length
122 * the length to set to, must be zero or positive
123 * @return this, to enable chaining
124 * @throws IndexOutOfBoundsException
125 * if the length is negative
126 */
127 public OpenStringBuilder setLength(final int length) {
128 if (length < 0) {
129 throw new StringIndexOutOfBoundsException(length);
130 }
131 if (length < size) {
132 size = length;
133 } else if (length > size) {
134 ensureCapacity(length);
135 final int oldEnd = size;
136 final int newEnd = length;
137 size = length;
138 for (int i = oldEnd; i < newEnd; i++) {
139 buffer[i] = '\0';
140 }
141 }
142 return this;
143 }

296 *
297 * @param startIndex
298 * the start index, inclusive, must be valid
299 * @param endIndex
300 * the end index, exclusive, must be valid except that if too large it is treated as end of string

307 final int len = endIndex - startIndex;

363 /**
364 * If possible, reads chars from the provided {@link Readable} directly into underlying character buffer without
365 * making extra copies.
366 *
367 * @param readable
368 * object to read from
369 * @return The number of characters read
370 * @throws IOException
371 * if an I/O error occurs
372 *
373 * @see #appendTo(Appendable)
374 */
375 public int readFrom(final Readable readable) throws IOException {
376 final int oldSize = size;
377 if (readable instanceof Reader) {
378 final Reader r = (Reader) readable;
379 ensureCapacity(size + 1);
380 int read;
381 while ((read = r.read(buffer, size, buffer.length - size)) != -1) {
382 size += read;
383 ensureCapacity(size + 1);
384 }
385 } else if (readable instanceof CharBuffer) {
386 final CharBuffer cb = (CharBuffer) readable;
387 final int remaining = cb.remaining();
388 ensureCapacity(size + remaining);
389 cb.get(buffer, size, remaining);
390 size += remaining;
391 } else {
392 while (true) {
393 ensureCapacity(size + 1);
394 final CharBuffer buf = CharBuffer.wrap(buffer, size, buffer.length - size);
395 final int read = readable.read(buf);
396 if (read == -1) {
397 break;
398 }
399 size += read;
400 }
401 }
402 return size - oldSize;
403 }
