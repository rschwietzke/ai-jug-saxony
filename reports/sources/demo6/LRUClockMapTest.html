42 /**
43 * The test suite for {@link LRUClockMap}.
44 *
45 * <p>Two groups of properties are under test here. The hash map part is exact and can be pinned
46 * down: what goes in comes out, the size is bounded by maxSize, no slot ever leaks and no probe
47 * chain ever gets a gap. The LRU part is deliberately approximate, the class says so itself, so it
48 * is tested for the guarantees the clock algorithm actually gives instead of for strict LRU order:
49 * an untouched entry dies within one turnover, a hot working set is kept, and a read marks an entry
50 * while {@code getRaw} does not.
51 *
52 * <p>One measured limit is worth knowing before reading L17 and L18: the protection a read buys only
53 * works from roughly 24 entries upwards. Below that the clock hand wraps around so quickly that it
54 * can clear an entry's flag and evict it in the same sweep, so on a map of 4 or 8 a hot entry is not
55 * kept at all. That is the "not really predictable" the class doc warns about, and the tests assert
56 * the strong property only for the sizes where it actually holds.
57 *
58 * @author René Schwietzke (Xceptance Software Technologies GmbH)
59 */
60 class LRUClockMapTest
61 {
62 /**
63 * Asserts the internal invariants. Cheap enough to call after every interesting step.
64 *
65 * <p>The chain check is the important one. The map probes linearly, so every entry must be
66 * reachable from its home position without running into a free slot on the way. A removal or an
67 * eviction that forgets to shift the followers back would break exactly this.
68 *
69 * @param map the map to check
70 * @param maxSize the maximum size the map was built with
71 */
72 private static <K, V> void assertHealthy(final LRUClockMap<K, V> map, final int maxSize)
73 {
74 assertEquals(map.size(), map.trueSize(), "size and the number of occupied slots differ");
75 assertTrue(map.size() <= maxSize, "the map holds more than maxSize entries");
76 assertEquals(1, Integer.bitCount(map.occupiedSpace()), "the capacity is not a power of two");
77 assertTrue(map.size() < map.occupiedSpace(), "the table is completely full, probing cannot terminate");
78
79 final List<LRUClockMap<K, V>.DebugWrapper<K, V>> slots = map.getDebugData();
80 final int length = slots.size();
81
82 for (int i = 0; i < length; i++)
83 {
84 final LRUClockMap<K, V>.DebugWrapper<K, V> w = slots.get(i);
85 if (w == null)
86 {
87 continue;
88 }
89
90 int ptr = w.truePosition;
91 while (ptr != i)
92 {
93 assertNotNull(slots.get(ptr), "the chain from " + w.truePosition + " to " + i + " has a gap at " + ptr);
94 ptr = (ptr + 1) % length;
95 }
96 }
97
98 // the keys the map reports must be exactly the occupied slots, and they must be unique
99 final List<K> keys = map.keys();
100 assertEquals(map.size(), keys.size(), "keys() does not match the size");
101 assertEquals(map.size(), new HashSet<>(keys).size(), "keys() contains duplicates");
102 }

105 * Reads the second chance flag of a key.
106 *
107 * @param map the map to look into
108 * @param key the key to look for, must be in the map
109 * @return the second chance flag of that entry
110 */
111 private static <K, V> boolean secondChanceOf(final LRUClockMap<K, V> map, final K key)
112 {
113 for (final LRUClockMap<K, V>.DebugWrapper<K, V> w : map.getDebugData())
114 {
115 if (w != null && w.key.equals(key))
116 {
117 return w.secondChance;
118 }
119 }
120
121 throw new AssertionError("Key " + key + " is not in the map");
122 }

128 * @param key the key to look for, must be in the map
129 * @return the home position of that entry

144 /**
145 * The length of the longest run of occupied slots. This is what the cost of a removal and of an
146 * eviction depends on, because repairing the probe chains has to walk the run behind the freed
147 * slot. A healthy table has short runs.
148 *
149 * @param map the map to measure
150 * @return the length of the longest cluster
151 */
152 private static <K, V> int longestCluster(final LRUClockMap<K, V> map)
153 {
154 final List<LRUClockMap<K, V>.DebugWrapper<K, V>> slots = map.getDebugData();
155 final int length = slots.size();
156
157 // start at a free slot, otherwise we would cut a wrapping cluster in two
158 int start = 0;
159 while (start < length && slots.get(start) != null)
160 {
161 start++;
162 }
163 if (start == length)
164 {
165 return length;
166 }
167
168 int longest = 0;
169 int current = 0;
170 for (int i = 1; i <= length; i++)
171 {
172 if (slots.get((start + i) % length) != null)
173 {
174 current++;
175 longest = Math.max(longest, current);
176 }
177 else
178 {
179 current = 0;
180 }
181 }
182
183 return longest;
184 }

189 * @param maxSize the size of the map to build

191 */
192 private static LRUClockMap<String, String> filled(final int maxSize)
193 {
194 final LRUClockMap<String, String> map = new LRUClockMap<>(maxSize);
195
196 for (int i = 0; i < maxSize; i++)
197 {
198 map.put("k" + i, "v" + i);
199 }
200
201 return map;
202 }

510 /**
511 * Builds a map that holds a hot set, an equally large cold set and some filler, then streams
512 * new keys through it while reading the hot set on every round.
513 *
514 * @param maxSize the size of the map to use
515 * @param rounds how many new keys to push through
516 * @return how many of the hot and how many of the cold entries survived
517 */
518 private int[] hotAndColdSurvivors(final int maxSize, final int rounds)
519 {
520 final LRUClockMap<String, String> map = new LRUClockMap<>(maxSize);
521 final int hotCount = maxSize / 4;
522
523 for (int i = 0; i < hotCount; i++)
524 {
525 map.put("hot" + i, "h" + i);
526 map.put("cold" + i, "c" + i);
527 }
528 for (int i = 0; i < maxSize - 2 * hotCount; i++)
529 {
530 map.put("filler" + i, "f" + i);
531 }
532
533 for (int round = 0; round < rounds; round++)
534 {
535 // touch the hot set before the insert, so the entries are marked when the
536 // eviction happens
537 for (int i = 0; i < hotCount; i++)
538 {
539 map.get("hot" + i);
540 }
541 map.put("new" + round, "n" + round);
542 }
543
544 int hot = 0;
545 int cold = 0;
546 for (int i = 0; i < hotCount; i++)
547 {
548 if (map.getRaw("hot" + i) != null)
549 {
550 hot++;
551 }
552 if (map.getRaw("cold" + i) != null)
553 {
554 cold++;
555 }
556 }
557
558 assertHealthy(map, maxSize);
559
560 return new int[] {hot, cold};
561 }

567 {

575 @ParameterizedTest(name = "L18 maxSize {0}")
576 @ValueSource(ints = {4, 8, 12, 16, 20})
577 @DisplayName("L18 on a small map the clock protects less, but never less than nothing")
578 void smallMapsGiveWeakerProtection(final int maxSize)
579 {
580 // This pins down the limit the class documents as "not really predictable". On a small
581 // map the hand needs only a few steps to wrap around, so it can clear an entry's flag
582 // and come back to evict it within the very same sweep, before the entry had a chance
583 // to be read again. Measured: up to a maxSize of 8 a hot entry gets no protection at
584 // all, from 24 upwards it is kept reliably, in between it is partial. So all a small
585 // map guarantees is that touching an entry never hurts.
586 final int[] survivors = hotAndColdSurvivors(maxSize, 2000);
587
588 assertTrue(survivors[0] >= survivors[1],
589 "touching entries made things worse, hot " + survivors[0] + " cold " + survivors[1]);
590 }

1151 // hashing as used by FastHashMap.homeSlot in this package.
