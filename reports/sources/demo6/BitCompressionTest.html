31 /**
32 * The test suite for {@link BitCompression}.
33 *
34 * <p>The two methods here are only ever used as a pair. Together they halve the resolution of a bit
35 * set that is used as a "which values did we see" sketch: bucket {@code 2k} and bucket {@code 2k+1}
36 * of the input become bucket {@code k} of the output, and a bucket is set in the output when either
37 * of its two sources was set. That composed behaviour is the real contract, so it gets its own
38 * exhaustive test over all 64 bit positions (T05) on top of the tests for the individual steps.
39 *
40 * <p>Note what the pair does <em>not</em> do: it never carries anything into the upper 32 bits. The
41 * output of a compression round always fits into bits 0..31. Callers that keep a 128 bucket sketch
42 * in two longs have to fold the compressed high word into the upper half of the low word themselves
43 * - see {@code IntTimeSeriesEntry} for one caller that does and one that forgets.
44 */
45 class BitCompressionTest
46 {
47 /**
48 * A fixed seed keeps the randomized tests reproducible.
49 */
50 private static final long SEED = 20260906L;
51
52 /**
53 * Applies the two steps the way every production caller does.
54 */
55 private static long compressionRound(final long value)
56 {
57 return BitCompression.compressAndShiftOddBits(BitCompression.combineAdjacentBits(value));
58 }
59
60 @Nested
61 @DisplayName("A. combineAdjacentBits")
62 class CombineAdjacentBits
63 {
64 @Test
65 @DisplayName("T01 a set bit is smeared into the next higher position")
66 void smearsUpwards()
67 {
68 assertEquals(0b11L, BitCompression.combineAdjacentBits(0b01L));
69 assertEquals(0b110L, BitCompression.combineAdjacentBits(0b10L));
70 assertEquals(0b111L, BitCompression.combineAdjacentBits(0b11L));
71 assertEquals(0L, BitCompression.combineAdjacentBits(0L));
72 }
73
74 @Test
75 @DisplayName("T02 the highest bit is shifted out, everything else survives")
76 void topBitFallsOff()
77 {
78 assertEquals(1L << 63, BitCompression.combineAdjacentBits(1L << 63), "nothing above bit 63 to smear into");
79 assertEquals((1L << 63) | (1L << 62), BitCompression.combineAdjacentBits(1L << 62));
80 assertEquals(-1L, BitCompression.combineAdjacentBits(-1L));
81 }
82
83 @Test
84 @DisplayName("T03 it is exactly v | (v << 1) for arbitrary words")
85 void matchesTheDefinition()
86 {
87 final Random random = new Random(SEED);
88 for (int i = 0; i < 10_000; i++)
89 {
90 final long value = random.nextLong();
91 assertEquals(value | (value << 1), BitCompression.combineAdjacentBits(value));
92 }
93 }
94 }
95
96 @Nested
97 @DisplayName("B. compressAndShiftOddBits")
98 class CompressAndShiftOddBits
99 {
100 @Test
101 @DisplayName("T04 the odd bits are packed into the lower 32 bits")
102 void packsOddBits()
103 {
104 assertEquals(0L, BitCompression.compressAndShiftOddBits(0L));
105 assertEquals(0xFFFFFFFFL, BitCompression.compressAndShiftOddBits(-1L), "all 32 odd bits set");
106 assertEquals(0xFFFFFFFFL, BitCompression.compressAndShiftOddBits(0xAAAAAAAAAAAAAAAAL), "only the odd bits matter");
107 assertEquals(0L, BitCompression.compressAndShiftOddBits(0x5555555555555555L), "even bits are dropped");
108
109 for (int bit = 0; bit < 64; bit++)
110 {
111 final long compressed = BitCompression.compressAndShiftOddBits(1L << bit);
112 final long expected = (bit % 2 == 0) ? 0L : (1L << (bit / 2));
113 assertEquals(expected, compressed, "input bit " + bit);
114 }
115 }
116
117 @Test
118 @DisplayName("T05 the result never uses the upper 32 bits")
119 void resultFitsIn32Bits()
120 {
121 final Random random = new Random(SEED);
122 for (int i = 0; i < 10_000; i++)
123 {
124 final long compressed = BitCompression.compressAndShiftOddBits(random.nextLong());
125 assertEquals(0L, compressed >>> 32, "the upper half must stay clean");
126 assertTrue(compressed >= 0, "and therefore the result is never negative");
127 }
128 }
129
130 @Test
131 @DisplayName("T06 it agrees with a straightforward bit by bit reference implementation")
132 void matchesReferenceImplementation()
133 {
134 final Random random = new Random(SEED);
135 for (int i = 0; i < 5_000; i++)
136 {
137 final long value = random.nextLong();
138
139 long expected = 0;
140 for (int bit = 1; bit < 64; bit += 2)
141 {
142 if ((value & (1L << bit)) != 0)
143 {
144 expected |= 1L << (bit / 2);
145 }
146 }
147
148 assertEquals(expected, BitCompression.compressAndShiftOddBits(value), "mismatch for " + Long.toHexString(value));
149 }
150 }
151 }

153 @Nested
154 @DisplayName("C. the composed compression round - what callers actually use")
155 class CompressionRound
156 {
157 @Test
158 @DisplayName("T07 bucket 2k and bucket 2k+1 both end up in bucket k")
159 void halvesTheResolution()
160 {
161 for (int bit = 0; bit < 64; bit++)
162 {
163 assertEquals(1L << (bit / 2), compressionRound(1L << bit), "input bit " + bit + " must land in bucket " + (bit / 2));
164 }
165 }
166
167 @Test
168 @DisplayName("T08 a round is lossy but never invents or loses a populated bucket")
169 void isLossyButComplete()
170 {
171 final Random random = new Random(SEED);
172 for (int i = 0; i < 5_000; i++)
173 {
174 final long value = random.nextLong();
175 final long compressed = compressionRound(value);
176
177 for (int bit = 0; bit < 64; bit++)
178 {
179 final boolean sourceSet = (value & (1L << bit)) != 0;
180 final boolean targetSet = (compressed & (1L << (bit / 2))) != 0;
181
182 if (sourceSet)
183 {
184 assertTrue(targetSet, "bucket " + (bit / 2) + " lost the value from bit " + bit);
185 }
186 }
187
188 for (int bit = 0; bit < 32; bit++)
189 {
190 if ((compressed & (1L << bit)) != 0)
191 {
192 final boolean anySourceSet = (value & ((1L << (2 * bit)) | (1L << (2 * bit + 1)))) != 0;
193 assertTrue(anySourceSet, "bucket " + bit + " was invented out of nothing");
194 }
195 }
196
197 assertTrue(Long.bitCount(compressed) <= Long.bitCount(value), "compression can only merge buckets, never split them");
198 }
199 }
200
201 @Test
202 @DisplayName("T09 repeated rounds converge to a single bucket")
203 void repeatedRoundsConverge()
204 {
205 long value = -1L;
206 for (int round = 1; round <= 6; round++)
207 {
208 value = compressionRound(value);
209 assertEquals(64 >> round, Long.bitCount(value), "after round " + round);
210 }
211
212 // six rounds turn 64 set buckets into one
213 assertEquals(1L, value);
214 assertEquals(1L, compressionRound(value), "bucket 0 stays bucket 0 forever");
215 assertEquals(0L, compressionRound(0L), "and an empty sketch stays empty");
216 }
217 }
