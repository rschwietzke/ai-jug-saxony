32 /**
33 * The test suite for {@link BitUtil}.
34 *
35 * <p>This is imported Apache Lucene/Solr code, so the interesting question is not "what should it
36 * do" but "does this copy still do what the JDK intrinsics do". Almost every test here is therefore
37 * a differential test against the JDK reference ({@link Long#bitCount},
38 * {@link Long#numberOfTrailingZeros}, {@link Long#numberOfLeadingZeros}) plus a handful of pinned
39 * corner cases the JDK and this class disagree on.
40 *
41 * <p>The {@code pop_*} family carries the bulk of the risk. Those methods process eight words per
42 * iteration with a carry-save-adder network and then handle the remaining words in a 4/2/1 cascade,
43 * so the word count alone decides which code path runs. The tests below sweep the word count from 0
44 * to 40 to walk through every combination of full blocks and tail branches instead of testing one
45 * arbitrary array.
46 *
47 * <p>Only {@link BitUtil#nextHighestPowerOfTwo(int)} is actually used by the production code in this
48 * module ({@code RuntimeHistogram} and {@code IntTimeSeries}), and its behaviour for 0, for negative
49 * input and above 2^30 is pinned here because both callers depend on it.
50 */
51 class BitUtilTest
52 {
53 /**
54 * A fixed seed keeps the randomized differential tests reproducible.
55 */
56 private static final long SEED = 20260906L;
57
58 /**
59 * Reference implementation for the pop_* family: count the bits of the words in range after
60 * applying the given combiner to the two arrays.
61 */
62 private static long reference(final long[] a, final long[] b, final int wordOffset, final int numWords,
63 final java.util.function.LongBinaryOperator op)
64 {
65 long sum = 0;
66 for (int i = wordOffset; i < wordOffset + numWords; i++)
67 {
68 sum += Long.bitCount(op.applyAsLong(a[i], b == null ? 0L : b[i]));
69 }
70
71 return sum;
72 }
73
74 /**
75 * Builds an array of pseudo random words. Every fourth word is forced to an extreme value so
76 * the carry-save-adder chain sees all-zero and all-one words too, not only dense random noise.
77 */
78 private static long[] randomWords(final Random random, final int length)
79 {
80 final long[] words = new long[length];
81 for (int i = 0; i < length; i++)
82 {
83 words[i] = switch (i % 4)
84 {
85 case 0 -> 0L;
86 case 1 -> -1L;
87 default -> random.nextLong();
88 };
89 }
90
91 return words;
92 }

254 @Test
255 @DisplayName("T12 the set identities hold: |a| + |b| = |a & b| + |a | b| and |a ^ b| = |a| + |b| - 2|a & b|")
256 void setIdentities()
257 {
258 final Random random = new Random(SEED);
259 final int length = 17;
260 final long[] a = randomWords(random, length);
261 final long[] b = randomWords(random, length);
262
263 final long popA = BitUtil.pop_array(a, 0, length);
264 final long popB = BitUtil.pop_array(b, 0, length);
265 final long and = BitUtil.pop_intersect(a, b, 0, length);
266 final long or = BitUtil.pop_union(a, b, 0, length);
267
268 assertEquals(popA + popB, and + or);
269 assertEquals(popA + popB - 2 * and, BitUtil.pop_xor(a, b, 0, length));
270 assertEquals(popA - and, BitUtil.pop_andnot(a, b, 0, length));
271 }
272 @Test
273 @DisplayName("T13 nearly saturated words drive the carry save adder into its eight word carry")
274 void saturatedWordsExerciseTheCarryChain()
275 {
276 // the accumulator only produces an 'eights' carry inside the trailing 4 and 2 word
277 // branches when almost every word is saturated. Random data practically never gets
278 // there, a run of all-ones words with a single hole in it does.
279 for (int length = 8; length <= 20; length++)
280 {
281 for (int hole = 0; hole < length; hole++)
282 {
283 final long[] a = new long[length];
284 final long[] b = new long[length];
285 Arrays.fill(a, -1L);
286 a[hole] = 0L;
287 b[(hole + 1) % length] = -1L;
288
289 final String message = "length " + length + ", hole at " + hole;
290 assertEquals(reference(a, null, 0, length, (x, __) -> x), BitUtil.pop_array(a, 0, length), message);
291 assertEquals(reference(a, b, 0, length, (x, y) -> x & y), BitUtil.pop_intersect(a, b, 0, length), message);
292 assertEquals(reference(a, b, 0, length, (x, y) -> x | y), BitUtil.pop_union(a, b, 0, length), message);
293 assertEquals(reference(a, b, 0, length, (x, y) -> x & ~y), BitUtil.pop_andnot(a, b, 0, length), message);
294 assertEquals(reference(a, b, 0, length, (x, y) -> x ^ y), BitUtil.pop_xor(a, b, 0, length), message);
295 }
