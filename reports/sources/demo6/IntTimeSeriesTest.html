33 /**
34 * The test suite for {@link IntTimeSeries}.
35 *
36 * <p>The class is a fixed size ring of per second buckets that never rejects data. It reacts to time
37 * stamps it cannot hold in two ways, and everything interesting in this suite is about those two
38 * reactions.
39 *
40 * <p><b>Condensing</b> happens when a time stamp arrives that is too far in the future for the
41 * current window. Every two neighbouring slots are merged into one, the slot width doubles, and the
42 * window covers twice as much wall clock time. The scale is an exponent plus one, so
43 * {@code getScale() == 1} means one second per slot, {@code 2} means two seconds, {@code 3} means
44 * four seconds - {@link IntTimeSeries#getSlotWidth()} is the readable version of the same thing.
45 *
46 * <p><b>Shifting</b> happens when a time stamp arrives before the current start of the window. The
47 * whole array is moved to the right to make room, which is the price for accepting out of order
48 * data.
49 *
50 * <p>Section G collects the cases where those two mechanisms break. They are real, reproducible and
51 * reachable with ordinary input - a large forward gap silently drops everything that was recorded
52 * before it, and a large backward gap throws an {@link ArrayIndexOutOfBoundsException} out of
53 * {@code addValue}. The tests pin the current behaviour on purpose, including the thresholds where
54 * it flips, so that a fix can be verified against them. {@code doc/XLT-DATA.md} describes what the
55 * replacement should do instead.
56 */

80 /**
81 * The number of slots that actually carry a sample.
82 */
83 private static int usedSlots(final IntTimeSeries series)
84 {
85 int used = 0;
86 for (final IntTimeSeriesEntry entry : series.getValues())
87 {
88 if (entry.getCount() > 0)
89 {
90 used++;
91 }
92 }
93
94 return used;
95 }
96
97 /**
98 * The sum over all slots, computed directly from the entries rather than through the statistics.
99 */
100 private static long sumOverSlots(final IntTimeSeries series)
101 {
102 long sum = 0;
103 for (final IntTimeSeriesEntry entry : series.getValues())
104 {
105 sum += entry.getTotalValue();
106 }
107
108 return sum;
109 }
